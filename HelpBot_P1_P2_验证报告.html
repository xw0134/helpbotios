@Nullable private static String getAnyAvailableTokenForFaq(@NonNull final Context context) { try { // 1) 内存 token（例如 pending login） final String pending = pendingLoginToken; if (pending != null && !pending.trim().isEmpty()) { return pending.trim(); } // 2) 加密存储 token（login 成功后写入） final EncryptedStorage storage = EncryptedStorage.getInstance(context.getApplicationContext()); if (storage != null && storage.isAvailable()) { final String stored = storage.getEncryptedString(TOKEN_STORAGE_KEY_JWT); if (stored != null && !stored.trim().isEmpty()) { return stored.trim(); } } return null; } catch (final Exception ignored) { return null; } }

  3. ✓ 异常处理完善，返回null时会触发NOT_LOGGED_IN错误
